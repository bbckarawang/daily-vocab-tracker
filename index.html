Google Sheets Backend Config

Leave empty to use Local Storage / Demo Mode.

How to Setup Google Sheets Backend (3 Steps):

  1. Create a new Google Sheet. Create two tabs: named Users and Submissions.
  2. Go to Extensions > Apps Script and paste the Google Apps Script code below.
  3. Click Deploy > New Deployment, set type to Web App, execute as Me, and access to Anyone. Copy the Web App URL and paste it above!
Google Apps Script Code (Code.gs): 
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var action = data.action;
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    if (action === "signup") {
      var userSheet = ss.getSheetByName("Users");
      userSheet.appendRow([data.email, data.password, data.name, new Date()]);
      return responseJSON({ status: "success", user: { email: data.email, name: data.name } });
    } 
    
    if (action === "login") {
      var userSheet = ss.getSheetByName("Users");
      var users = userSheet.getDataRange().getValues();
      for (var i = 1; i < users.length; i++) {
        if (users[i][0] === data.email && users[i][1] === data.password) {
          return responseJSON({ status: "success", user: { email: users[i][0], name: users[i][2] } });
        }
      }
      return responseJSON({ status: "error", message: "Invalid email or password" });
    }
    
    if (action === "addVocab") {
      var subSheet = ss.getSheetByName("Submissions");
      subSheet.appendRow([
        data.userEmail, 
        data.date, 
        data.vocab, 
        data.category, 
        data.sentence, 
        data.isLate ? "LATE" : "ON_TIME", 
        new Date()
      ]);
      return responseJSON({ status: "success" });
    }
    
    if (action === "getData") {
      var subSheet = ss.getSheetByName("Submissions");
      var rows = subSheet.getDataRange().getValues();
      var userVocabs = [];
      for (var j = 1; j < rows.length; j++) {
        if (rows[j][0] === data.userEmail) {
          userVocabs.push({
            date: rows[j][1],
            vocab: rows[j][2],
            category: rows[j][3],
            sentence: rows[j][4],
            status: rows[j][5]
          });
        }
      }
      return responseJSON({ status: "success", vocabs: userVocabs });
    }

  } catch(err) {
    return responseJSON({ status: "error", message: err.toString() });
  }
}

function responseJSON(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
